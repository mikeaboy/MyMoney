💡 30 秒免費建立您的私人 Google Drive 自動同步端點：

1. 前往 script.google.com 點擊「新專案」，貼上下方程式碼。

2. 點擊右上角「部署」➔「新增部署作業」➔ 類型選「網頁應用程式」，執行身分選「我」，誰可以存取選「任何人」，複製產生的網址貼回上方即可永久自動同步於您個人的 Google Drive（檔案名稱：wealth_manager_sync.json）！

const FILE_NAME = 'wealth_manager_sync.json';
function doGet() {
  const files = DriveApp.getFilesByName(FILE_NAME);
  const content = files.hasNext() ? files.next().getBlob().getDataAsString() : '{}';
  return ContentService.createTextOutput(content).setMimeType(ContentService.MimeType.JSON);
}
function doPost(e) {
  const files = DriveApp.getFilesByName(FILE_NAME);
  if (files.hasNext()) files.next().setContent(e.postData.contents);
  else DriveApp.createFile(FILE_NAME, e.postData.contents, MimeType.PLAIN_TEXT);
  return ContentService.createTextOutput('{"ok":true}').setMimeType(ContentService.MimeType.JSON);
}
